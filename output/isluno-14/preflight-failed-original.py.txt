def require(ok,code):
    if not ok:raise Stop(code)

def run(argv,cap=2097152):
    left=110-(time.monotonic()-START);require(left>0,'preflight_deadline')
    p=subprocess.run(argv,stdin=subprocess.DEVNULL,stdout=subprocess.PIPE,stderr=subprocess.PIPE,timeout=min(12,left),check=False)
    require(len(p.stdout)+len(p.stderr)<=cap,'command_output_bound');require(p.returncode==0,'selected_command_failed')
    return p.stdout.decode()

def read(path,cap=2097152):
    p=Path(path);require(p.is_absolute(),'absolute_path')
    for a in [*reversed(p.parents),p]:require(not a.is_symlink(),'selected_symlink')
    fd=os.open(p,os.O_RDONLY|os.O_NOFOLLOW)
    try:
        s=os.fstat(fd);require(stat.S_ISREG(s.st_mode) and s.st_size<=cap,'selected_file_shape')
        raw=os.read(fd,cap+1);require(len(raw)<=cap,'selected_file_bound');return raw
    finally:os.close(fd)

def sha(raw):return hashlib.sha256(raw).hexdigest()
